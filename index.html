Enlace de Google Apps Script

Pega aquí la URL que te dio Google Apps Script al publicar el proyecto como aplicación web. Solo se hace una vez.

¿No conecta? Cómo configurar el script
  1. En tu hoja de Sheets: Extensiones → Apps Script, y pega el código de abajo.
  2. Implementar → Nueva implementación → tipo "Aplicación web".
  3. "Ejecutar como": Yo. "Quién tiene acceso": Cualquier usuario.
  4. Copia la URL que termina en /exec (no la de /dev) y pégala arriba.
  5. Si vuelves a editar el código, debes crear una nueva versión en Gestionar implementaciones — si no, la URL sigue usando el código viejo.
  6. Los egresos y la caja chica se guardan en sus propias pestañas ("Egresos" y "CajaChica") de la misma hoja.
function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const d = JSON.parse(e.postData.contents);
  if (d.tipo === 'egreso') {
    const sh = ss.getSheetByName('Egresos') || ss.insertSheet('Egresos');
    sh.appendRow([d.timestamp, d.fecha, d.vendedor, d.monto, d.concepto]);
  } else if (d.tipo === 'caja_chica') {
    const sh = ss.getSheetByName('CajaChica') || ss.insertSheet('CajaChica');
    sh.appendRow([d.timestamp, d.fecha, d.vendedor, d.monto]);
  } else {
    const sh = ss.getSheetByName('Ventas') || ss.insertSheet('Ventas');
    sh.appendRow([d.timestamp, d.fecha, d.hour24, d.vendedor, d.total, d.paid, d.change]);
  }
  return ContentService.createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Ventas');
  if (!sh) return ContentService.createTextOutput('[]')
    .setMimeType(ContentService.MimeType.JSON);
  const rows = sh.getDataRange().getValues().slice(1);
  const sales = rows.map(r => ({
    timestamp:r[0], fecha:r[1], hour24:r[2], vendedor:r[3],
    total:r[4], paid:r[5], change:r[6]
  }));
  return ContentService.createTextOutput(JSON.stringify(sales))
    .setMimeType(ContentService.MimeType.JSON);
}
